[31m+ export async function searchApprovedKnowledge([39m
[31m+   userId: number,[39m
[31m+   organisationId: number,[39m

[31m+ ) {[39m

[31m+       (total, term) => total + (haystack.includes(term) ? 1 : 0),[39m
[31m+       0[39m
[31m+     );[39m
[31m+   };[39m
[31m+   return sources[39m
[31m+     .map(source => ({ source, score: score(source) }))[39m
[31m+     .filter(item => item.score > 0)[39m
